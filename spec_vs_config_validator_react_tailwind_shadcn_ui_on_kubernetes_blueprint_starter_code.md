# Spec vs Config Validator — Implementation Blueprint

> Frontend: **React + TypeScript + Tailwind + shadcn/ui**  
> Backend: **FastAPI (Python)**  
> Packaging: **Docker / OCI**  
> Platform: **Kubernetes** (Ingress + HPA + RBAC + PVC)  
> Focus: 可上傳客戶規格表（Spec）與設備組態（Config）並自動比對，輸出報告，支援離線與企業內部環境。

## 2) 資料流（Data Flow）
1. 使用者上傳 **Spec.xlsx/CSV** 與 **Config.txt/REST JSON** → `POST /api/specs:upload`, `POST /api/configs:upload`。
2. API 以 **YAML Parser** 將 CLI 文字抽取為標準化 JSON（Cluster/Node/BMC/Version/Image/Aggregate/Disk/Port）。
3. 規則引擎以 **rulesets/default.yaml** 比對 → 產生差異（P0/P1/P2）與摘要。
4. 前端呈現 **Dashboard**、**Diff Table**、**建議修正**，可匯出 PDF/CSV/JSON。

### 4.2 上傳精靈（UploadWizard）
```tsx
// apps/web/src/components/UploadWizard.tsx
import { useState } from 'react';
import { Button } from '@/components/ui/button';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';

export default function UploadWizard({ onUploaded }: { onUploaded: (jobId: string)=>void }) {
  const [specFile, setSpecFile] = useState<File|null>(null);
  const [confFiles, setConfFiles] = useState<FileList|null>(null);
  const [busy, setBusy] = useState(false);

  const upload = async () => {
    setBusy(true);
    const specFd = new FormData();
    if (specFile) specFd.append('file', specFile);
    const specResp = specFile ? await fetch('/api/specs:upload', { method:'POST', body: specFd }) : null;

    const confFd = new FormData();
    if (confFiles) Array.from(confFiles).forEach(f => confFd.append('files', f));
    const confResp = await fetch('/api/configs:upload', { method:'POST', body: confFd });

    const { jobId } = await confResp.json();
    setBusy(false);
    onUploaded(jobId);
  };

  return (
    <Card className="shadow-sm">
      <CardHeader>
        <CardTitle>Upload Spec & Config</CardTitle>
      </CardHeader>
      <CardContent className="space-y-4">
        <div>
          <label className="text-sm font-medium">Spec (Excel/CSV/JSON)</label>
          <input type="file" accept=".xlsx,.csv,.json" onChange={e=>setSpecFile(e.target.files?.[0]||null)} className="block mt-1"/>
        </div>
        <div>
          <label className="text-sm font-medium">Config (one or more .txt/.json)</label>
          <input multiple type="file" accept=".txt,.log,.json" onChange={e=>setConfFiles(e.target.files)} className="block mt-1"/>
        </div>
        <Button disabled={busy} onClick={upload}>{busy? 'Uploading…':'Start Validate'}</Button>
      </CardContent>
    </Card>
  );
}
```

function format(v:any){
  if (v==null) return '-';
  if (typeof v==='object') return <pre className="text-xs whitespace-pre-wrap">{JSON.stringify(v,null,2)}</pre>;
  return String(v);
}

### 4.4 驗證主頁（Validate）

// apps/web/src/pages/Validate.tsx
import { useState } from 'react';
import UploadWizard from '@/src/components/UploadWizard';
import DiffTable from '@/src/components/DiffTable';
import type { ValidationResponse } from '@/src/lib/types';

export default function Validate(){
  const [result, setResult] = useState<ValidationResponse|null>(null);

          <DiffTable rows={result.details} />
        </div>

    </div>

}

class Bmc(BaseModel):
    ip: str; netmask: str; gateway: str; fw: Optional[str] = None

    current: str; default: Optional[str] = None; installedAt: Optional[str] = None

**app/main.py（精簡）**
```py
from fastapi import FastAPI, UploadFile, File, Query
from fastapi.middleware.cors import CORSMiddleware
from .schemas import ValidationResponse
from .services.parse_service import parse_spec, parse_configs
from .services.validate_service import validate

app = FastAPI(title='Spec vs Config API', version='0.1.0')
app.add_middleware(CORSMiddleware, allow_origins=['*'], allow_credentials=True, allow_methods=['*'], allow_headers=['*'])

# naive in-memory job store for MVP
JOBS: dict[str, dict] = {}

@app.post('/specs:upload')
async def upload_spec(file: UploadFile = File(...)):
    spec = await parse_spec(file)
    job_id = 'job-spec'
    JOBS.setdefault(job_id, {})['spec'] = spec
    return { 'jobId': job_id }

@app.post('/configs:upload')
async def upload_configs(files: list[UploadFile] = File(...)):
    conf = await parse_configs(files)
    job_id = 'job-spec'  # tie to same job for MVP
    JOBS.setdefault(job_id, {})['conf'] = conf
    return { 'jobId': job_id }

@app.post('/validate', response_model=ValidationResponse)
async def run_validate(jobId: str = Query(...)):
    job = JOBS.get(jobId)
    assert job and 'spec' in job and 'conf' in job, 'spec/conf missing'
    return validate(job['spec'], job['conf'])
```

**app/services/parse_service.py（示意）**
```py
import re, yaml
from fastapi import UploadFile

async def parse_spec(file: UploadFile):
    # TODO: map Excel/CSV to canonical schema; MVP accept JSON
    if file.filename.endswith('.json'):
        return yaml.safe_load((await file.read()).decode())
    return { 'cluster': { 'name': 'Unknown', 'expected_version': '9.11.1P16' }, 'nodes': [] }

async def parse_configs(files: list[UploadFile]):
    text = (await files[0].read()).decode(errors='ignore')
    # Very simple examples:
    version = re.search(r"NetApp Release\s+([\w\.]+)", text)
    bmc = re.search(r"IPv4\s+(\d+\.\d+\.\d+\.\d+)\s+(\d+\.\d+\.\d+\.\d+)\s+(\d+\.\d+\.\d+\.\d+)", text)
    return {
        'cluster': { 'name': 'from-config', 'version': version.group(1) if version else None },
        'nodes': [{ 'name':'node-1', 'bmc': { 'ip': bmc.group(1), 'netmask': bmc.group(2), 'gateway': bmc.group(3) } if bmc else None }]
    }
```

**app/services/validate_service.py（示意）**
```py
from . import rulesets

def validate(spec, conf):
    details = []
    # Example P0: ONTAP version equals or higher patch
    exp = (spec.get('cluster') or {}).get('expected_version')
    act = (conf.get('cluster') or {}).get('version')
    status = 'PASS' if act == exp else 'FAIL'
    details.append({ 'id':'R-002','level':'P0','title':'ONTAP version','status':status,'path':'cluster.version','expected':exp,'actual':act })

    # Example P0: BMC IP
    exp_bmc = (((spec.get('nodes') or []) + [{}])[0] or {}).get('bmc', {})
    act_bmc = (((conf.get('nodes') or []) + [{}])[0] or {}).get('bmc', {})
    for f in ('ip','netmask','gateway'):
        e, a = exp_bmc.get(f), act_bmc.get(f)
        details.append({ 'id':f'R-001-{f}','level':'P0','title':f'BMC {f}','status':'PASS' if e==a else 'FAIL','path':f'nodes[0].bmc.{f}','expected':e,'actual':a })

    p = sum(1 for d in details if d['status']=='PASS')
    summary = { 'passRate': p/len(details) if details else 1.0, 'p0': len([d for d in details if d['level']=='P0']), 'p1': 0, 'p2': 0 }
    return { 'summary': summary, 'details': details }
```

## 9) 安全與治理（要點）
- **CORS**：Ingress 層或 API 層鎖定允許來源；預設僅 web 網域。
- **上傳大小限制**：Ingress/NGINX 及 API 層同步設 `client_max_body_size`（例如 50MB）。
- **認證**：OIDC（Azure AD/Okta）— 前端以 PKCE，API 驗 JWT；驗證角色（Viewer/Editor/Approver）。
- **審計**：記錄每次上傳、解析後 JSON、規則版本、使用者與時間戳；存放於關聯式資料庫（PostgreSQL）。
- **機敏資料**：不存密碼；報告匯出不包含機敏欄位；需要時以 KMS/Secrets 管理。

## 11) CI/CD 指南（摘要）
1. **Build**：Node 20 + `npm ci && npm run build`；Python 3.12 + `pip install -r`。

3. **Test**：Web（Vitest + React Testing Library）、API（Pytest）。
4. **Image**：`docker buildx bake` 多平台；標記 `:gitsha` 與 `:semver`。
5. **Deploy**：Kustomize/Helm（二擇一）；本稿提供原生 YAML。
